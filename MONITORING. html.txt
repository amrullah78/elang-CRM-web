        const GAS_WEB_APP_URL = 'GANTI_DENGAN_URL_DEPLOYMENT_APPS_SCRIPT_ANDA';
        
        // Variabel global untuk menyimpan data yang diambil dari server
        let salesmen = [];
        let outlets = [];

        // --- Helper Functions: Komunikasi dengan Google Apps Script (GAS) ---
        async function fetchDataFromGas(sheetName) {
            try {
                const url = `${GAS_WEB_APP_URL}?sheet=${sheetName}`;
                const response = await fetch(url);
                const result = await response.json();
                
                if (result.status === 'success') {
                    return result.data;
                } else {
                    throw new Error(result.message || "Gagal memuat data dari Apps Script.");
                }
            } catch (error) {
                console.error("Error fetching data:", error);
                showNotification("Error Server", `Gagal memuat data ${sheetName}. Pastikan URL Apps Script sudah benar dan ter-deploy. Detail: ${error.message}`, false);
                return [];
            }
        }

        async function sendDataToGas(sheetName, data) {
            try {
                const url = `${GAS_WEB_APP_URL}?sheet=${sheetName}`;
                const response = await fetch(url, {
                    method: 'POST',
                    body: JSON.stringify(data),
                    headers: {
                        'Content-Type': 'text/plain;charset=utf-8' 
                    }
                });
                const result = await response.json();
                
                if (result.status === 'success') {
                    return true;
                } else {
                    throw new Error(result.message || "Gagal mengirim data ke Apps Script.");
                }
            } catch (error) {
                console.error("Error sending data:", error);
                showNotification("Error Server", `Gagal menyimpan data ${sheetName}. Detail: ${error.message}`, false);
                return false;
            }
        }

        async function deleteDataFromGas(sheetName, id) {
            try {
                // Apps Script hanya mendukung GET/POST, kita akan menggunakan GET dengan parameter 'method=DELETE'
                const url = `${GAS_WEB_APP_URL}?sheet=${sheetName}&id=${id}&method=DELETE`; 
                const response = await fetch(url);
                const result = await response.json();
                
                if (result.status === 'success') {
                    return true;
                } else {
                    throw new Error(result.message || "Gagal menghapus data.");
                }
            } catch (error) {
                console.error("Error deleting data:", error);
                showNotification("Error Server", `Gagal menghapus data. Detail: ${error.message}`, false);
                return false;
            }
        }

        async function initialLoadMasterData() {
            salesmen = await fetchDataFromGas('Salesmen');
            outlets = await fetchDataFromGas('Outlets');
            
            salesmen.sort((a, b) => a.nama_salesman.localeCompare(b.nama_salesman));

                option.value = s.nama_salesman; 
                option.textContent = s.nama_salesman;
                salesmanSelect.appendChild(option);

                const option = document.createElement('option');
                option.value = o.nama_outlet;

            });
        }

        // Fungsi Impor (Logika sama, hanya diperpendek)
        async function processImport(file, sheetName, requiredHeaders) {
            return new Promise((resolve, reject) => {
                if (!file) { reject("Tidak ada file dipilih."); return; }
                const reader = new FileReader();
                reader.onload = async function(e) {
                    try {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const worksheet = workbook.Sheets[workbook.SheetNames[0]];
                        const jsonData = XLSX.utils.sheet_to_json(worksheet, { raw: false, header: 1 });
                        
                        if (jsonData.length < 2) { reject("File kosong atau hanya berisi header."); return; }

                        const fileHeaders = jsonData[0].map(h => h ? h.toString().trim() : '');
                        const headerMap = requiredHeaders.map(req => {
                            const index = fileHeaders.findIndex(h => h.toLowerCase().replace(/\s/g, '') === req.toLowerCase().replace(/\s/g, ''));
                            if (index === -1) throw new Error(`Kolom header wajib: '${req}' tidak ditemukan.`);
                            return { requiredName: req, columnIndex: index };
                        });

                        let importedCount = 0;
                        let skippedCount = 0;
                        for (let i = 1; i < jsonData.length; i++) {
                            const row = jsonData[i];
                            const dataToSend = {};
                            headerMap.forEach(map => {
                                dataToSend[map.requiredName] = row[map.columnIndex] ? row[map.columnIndex].toString().trim() : '';
                            });
                            
                            let isSkipped = false;
                            if (sheetName === 'Salesmen' && salesmen.some(s => s.nama_salesman.toLowerCase() === dataToSend['Nama Salesman'].toLowerCase())) {
                                isSkipped = true;
                            } else if (sheetName === 'Outlets' && outlets.some(o => o.nama_outlet.toLowerCase() === dataToSend['Nama Outlet'].toLowerCase())) {
                                isSkipped = true;
                            }

                            if (!isSkipped) {
                                if (await sendDataToGas(sheetName, dataToSend)) {
                                    importedCount++;
                                } else {
                                    skippedCount++;
                                }
                            } else {
                                skippedCount++;
                            }
                        }
                        
                        await initialLoadMasterData(); 
                        resolve({ importedCount, skippedCount });

                    } catch (error) {
                        reject(error.message);
                    }
                };
                reader.onerror = () => reject("Gagal membaca file.");
                reader.readAsArrayBuffer(file);
            });
        }

        function populatePlanFilters() {
            const areaFilter = document.getElementById('planAreaFilter');
            const typeFilter = document.getElementById('planTypeFilter');

            const uniqueAreas = [...new Set(outlets.map(o => o.area_outlet))].filter(Boolean);
            const uniqueTypes = [...new Set(outlets.map(o => o.type_outlet))].filter(Boolean);

            areaFilter.length = 1;
            typeFilter.length = 1;

            uniqueAreas.sort().forEach(area => {
                const option = document.createElement('option');
                option.value = area;
                option.textContent = area;
                areaFilter.appendChild(option);
            });

            uniqueTypes.sort().forEach(type => {
                const option = document.createElement('option');
                option.value = type;
                option.textContent = type;
                typeFilter.appendChild(option);
            });
        }

        function renderPlanList() {
            const selectedArea = document.getElementById('planAreaFilter').value;
            const selectedType = document.getElementById('planTypeFilter').value;
            const outletListContainer = document.getElementById('planOutletList');
            const noDataMsg = document.getElementById('planNoDataMessage');
            outletListContainer.innerHTML = '';

            const filteredOutlets = outlets.filter(outlet => {
                const areaMatch = selectedArea === 'all' || outlet.area_outlet === selectedArea;
                const typeMatch = selectedType === 'all' || outlet.type_outlet === selectedType;
                return areaMatch && typeMatch;
            });
            
            if (filteredOutlets.length === 0) {
                noDataMsg.style.display = 'block';
                return;
            }
            noDataMsg.style.display = 'none';

            filteredOutlets.forEach(outlet => {
                const outletCard = document.createElement('div');
                outletCard.className = 'border p-4 rounded-lg bg-gray-50 shadow-sm';
                outletCard.innerHTML = `
                    <h4 class="font-bold text-md text-blue-700">${outlet.nama_outlet}</h4>

                    <div class="mt-2 flex justify-between items-center text-xs">

                        <span class="bg-green-100 text-green-800 px-2 py-1 rounded-full font-medium">${outlet.type_outlet}</span>
                    </div>
                `;
                outletListContainer.appendChild(outletCard);

        function formatRupiah(number) {
            if (typeof number === 'number') {
                number = number.toString();
            }
            if (!number) return '';
            let cleanNumber = number.replace(/[^0-9]/g, '').toString();
            let rupiah = cleanNumber.replace(/\B(?=(\d{3})+(?!\d))/g, ".");
            return rupiah ? 'Rp ' + rupiah : '';
        }

        function parseRupiah(rupiahString) {
            if (!rupiahString) return 0;
            return parseInt(rupiahString.replace(/Rp\s?|\./g, '').replace(/,/g, '')) || 0;
        }

            const value = this.value;

        });

            const nominalOrder = formatRupiah(parseRupiah(orderRaw));

                // Key harus sama persis dengan Header di Google Sheet
                'ID': 'V' + Date.now(), 
                'Tanggal Kunjungan': document.getElementById('tanggalKunjungan').value,
                'Salesman': document.getElementById('namaSalesman').value,
                'Outlet': document.getElementById('namaOutlet').value,
                'Type Outlet': document.getElementById('typeOutlet').value,
                'Alamat Outlet': document.getElementById('alamatOutlet').value,
                'Area': document.getElementById('area').value,
                'Status Transaksi': statusTransaksi,
                'Order': nominalOrder, 
                'Keterangan': document.getElementById('keterangan').value
            };

            if (await sendDataToGas('Visits', newVisit)) {
                await initialLoadVisits(); 
                renderVisitTable();
                showNotification("Sukses", "Data kunjungan berhasil dicatat dan disinkronkan!");
                this.reset();

                document.getElementById('alamatOutlet').value = '';

                document.getElementById('order').removeAttribute('required');
                document.getElementById('order').setAttribute('placeholder', 'Isi jika status "Order"');
                document.getElementById('tanggalKunjungan').valueAsDate = new Date(); 
            }
        });
        
        function renderVisitTable() {
            const tableBody = document.getElementById('visitTableBody');
            const noDataMsg = document.getElementById('noDataMessage');
            tableBody.innerHTML = '';
            
            if (visits.length === 0) {
                noDataMsg.style.display = 'block';
                return;
            }
            noDataMsg.style.display = 'none';

            visits.forEach((visit, index) => {
                const row = tableBody.insertRow();
                
                const statusClass = 
                    visit.status_transaksi === 'Order' ? 'bg-green-100 text-green-800' :
                    visit.status_transaksi === 'Tidak Order' ? 'bg-red-100 text-red-800' :
                    visit.status_transaksi === 'Batal Kunjungan' ? 'bg-gray-100 text-gray-800' :
                    'bg-yellow-100 text-yellow-800';
                
                row.innerHTML = `
                    <td class="text-center">${index + 1}</td>
                    <td>${visit.tanggal_kunjungan}</td>
                    <td>${visit.salesman}</td>
                    <td>${visit.outlet}</td>
                    <td>${visit.type_outlet}</td>
                    <td>${visit.alamat_outlet}</td>
                    <td>${visit.area}</td>
                    <td><span class="inline-block px-2 py-1 text-xs font-semibold rounded ${statusClass}">${visit.status_transaksi}</span></td>
                    <td class="whitespace-nowrap text-right">${visit.order || '-'}</td>
                    <td class="max-w-xs overflow-hidden text-ellipsis whitespace-nowrap">${visit.keterangan || '-'}</td>
                    <td>
                        <button onclick="openDeleteModal('${visit.id}')" class="text-red-600 hover:text-red-900 focus:outline-none">
                            <svg xmlns="http://www.w3.org/2000/svg" class="h-5 w-5" viewBox="0 0 20 20" fill="currentColor"><path fill-rule="evenodd" d="M9 2a1 1 0 00-.894.553L7.382 4H4a1 1 0 000 2v10a2 2 0 002 2h8a2 2 0 002-2V6a1 1 0 100-2h-3.382l-.724-1.447A1 1 0 0011 2H9zM7 8a1 1 0 012 0v6a1 1 0 11-2 0V8zm4 0a1 1 0 100 2v6a1 1 0 100-2V8z" clip-rule="evenodd" /></svg>
